    this.scheduleDate,
    this.scheduleTime,
    this.action,
    this.notes,
    this.dependentId,
    this.takenAt,
  });

  bool get isNext => action == 'NEXT';
  bool get isTaken => action == 'Taken';
  bool get isSkipped => action == 'Skipped';

  DateTime? get scheduledDateTime {
    if (scheduleDate == null || scheduleTime == null) return null;
    final parts = scheduleTime!.split(':');
    if (parts.length < 2) return null;
    final hour = int.tryParse(parts[0]) ?? 0;
    final minute = int.tryParse(parts[1]) ?? 0;
    return DateTime(
      scheduleDate!.year,
      scheduleDate!.month,
      scheduleDate!.day,
      hour,
      minute,
    );
  }

  /// Garante que existe um agendamento NEXT para o tratamento.
  Future<void> ensureNextScheduleForTreatment(Treatment treatment) async {
    if (!treatment.isActive || treatment.isRescueTreatment) return;
    if (treatment.endDate != null && treatment.endDate!.isBefore(DateTime.now())) return;

    final nextDose = await calculateNextDoseTime(treatment);
    if (nextDose == null) return;

    // Remove agendamentos NEXT antigos para este tratamento
    await _scheduleRepository.deleteNextSchedulesForTreatment(treatment.id);

    final scheduleDate = DateTime(nextDose.year, nextDose.month, nextDose.day);
    final scheduleTime =
        '${nextDose.hour.toString().padLeft(2, '0')}:${nextDose.minute.toString().padLeft(2, '0')}:00';

    final schedule = Schedule(
      id: DateTime.now().millisecondsSinceEpoch.toString(),
      createdAt: DateTime.now(),
      scheduleDate: scheduleDate,
      scheduleTime: scheduleTime,
      action: 'NEXT',
      dependentId: treatment.dependentId,
    );

    final treatmentInSchedule = TreatmentInSchedule(
      id: DateTime.now().millisecondsSinceEpoch.toString(),
      treatmentId: treatment.id,
      medicationId: treatment.medicationId,

    // Agenda notificação local
    final medication = treatment.medicationId != null
        ? await _medicationRepository.getMedicationById(treatment.medicationId!)
        : null;

  Future<DateTime?> calculateNextDoseTime(Treatment treatment) async {
    if (treatment.scheduleType == 'weekly') {
      return _calculateWeeklyNextDose(treatment);
    }
    return _calculatePeriodicNextDose(treatment);
  }

  DateTime? _calculatePeriodicNextDose(Treatment treatment) {
    if (treatment.firstDoseTime == null || treatment.frequencyHours <= 0) return null;

    final now = DateTime.now();
    final startDate = treatment.startDate;
    final firstDose = treatment.firstDoseTime!;

    var nextDose = DateTime(
      startDate.year,
      startDate.month,
      startDate.day,
      firstDose.hour,
      firstDose.minute,
    );

    // Se já houver doses tomadas/puladas, começa a partir da última ação
    // Para simplificar, usa o algoritmo básico: avança de frequency_hours até passar de now
    while (nextDose.isBefore(now) || nextDose.isAtSameMomentAs(now)) {
      nextDose = nextDose.add(Duration(hours: treatment.frequencyHours));
    }

    return nextDose;
  }

  Future<DateTime?> _calculateWeeklyNextDose(Treatment treatment) async {
    // Busca horários semanais do tratamento
    final response = await Supabase.instance.client
        .from('medication_times_on_treatment')
        .select()
        .eq('treatment_id', treatment.id)
        .order('time');

    final times = response
        .map((json) => MedicationTimeOnTreatment(
              id: json['id'] as String,
              treatmentId: json['treatment_id'] as String,
              dayOfWeek: json['day_of_week'] as int,
              time: _parseTime(json['time'] as String),
              dosage: (json['dosage'] as num).toDouble(),
              createdAt: DateTime.parse(json['created_at'] as String),
            ))
        .toList();

    if (times.isEmpty) return null;

    final now = DateTime.now();
    for (var i = 0; i < 14; i++) {
      final currentDate = now.add(Duration(days: i));
      final dayOfWeek = currentDate.weekday % 7;
      final dayTimes = times.where((t) => t.dayOfWeek == dayOfWeek).toList();
      for (final time in dayTimes) {

          currentDate.year,
          currentDate.month,
          currentDate.day,
          time.time.hour,
          time.time.minute,
        );
        if (doseTime.isAfter(now)) {
          return doseTime;
        }
      }
    }
    return null;
